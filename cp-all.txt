import { useFirestore } from "@/firebase"

import { useCaseAccess, useMyCaseSheets, deleteCaseSheet } from "@/hooks/use-case-sheets"

export default function CasePresentationHub() {
  const { user, allowed, loading } = useCaseAccess()
  const { sheets, loading: sheetsLoading } = useMyCaseSheets()
  const db = useFirestore()
  const [open, setOpen] = useState<string | null>("medicine")
  const [search] = useTopSearch("Search cases…")
  const q = search.trim().toLowerCase()

  if (loading) return <div className="flex justify-center py-24"><Loader2 className="h-8 w-8 animate-spin text-primary" /></div>

  const remove = async (id: string) => {
    if (!db || !user) return
    if (!window.confirm("Delete this saved case?")) return
    await deleteCaseSheet(db, user.uid, id)
  }

  return (
    <div className="max-w-5xl mx-auto px-4 py-8 space-y-8">
      <div>
        <h1 className="text-3xl font-bold tracking-tight">Case <span className="text-accent">Presentation</span></h1>
        <p className="text-sm text-muted-foreground mt-1 max-w-2xl">
          Pick a subject and case type. The proforma questions are already there: tap the options that apply, type anything extra, then press <b>Present case</b> for a professional write-up you can print or present.
        </p>
        {!allowed && (
          <div className="mt-4 rounded-2xl border border-accent/30 bg-accent/10 p-4 flex items-center gap-3 flex-wrap">
            <Lock className="h-5 w-5 text-accent" />
            <div className="text-sm flex-1 min-w-[200px]">Case Presentation is available to subscribers. Scholar is ₹999/year.</div>
            <Link href="/pricing"><Button size="sm">View plans</Button></Link>
          </div>
        )}
      </div>

      {user && allowed && (
        <section className="space-y-3">
          <h2 className="text-lg font-bold">My saved cases</h2>
          {sheetsLoading ? <Loader2 className="h-5 w-5 animate-spin text-primary" /> : sheets.length === 0 ? (
            <p className="text-sm text-muted-foreground">Nothing saved yet. Start a case below: it saves automatically.</p>
          ) : (
            <div className="grid gap-3 sm:grid-cols-2">
              {sheets.map((s) => {
                const c = getCase(s.caseType)
                return (
                  <div key={s.id} className="glass rounded-2xl border border-white/10 p-4 flex items-center gap-3">
                    <Link href={`/case-presentation/${s.caseType}/${s.id}`} className="flex-1 min-w-0">
                      <div className="font-semibold truncate">{s.title || "Untitled case"}</div>
                      <div className="text-xs text-muted-foreground truncate">{c?.title ?? s.caseType}{s.updatedAt?.toDate ? ` · ${s.updatedAt.toDate().toLocaleDateString()}` : ""}</div>

                  </div>
                )
              })}
            </div>
          )}
        </section>
      )}

      <section className="space-y-3">
        <h2 className="text-lg font-bold">Start a new case</h2>
        <div className="space-y-3">
          {SUBJECTS.map((s) => {
            const cases = casesFor(s.id).filter((c) => !q || `${c.title} ${c.blurb}`.toLowerCase().includes(q))
            if (q && cases.length === 0) return null
            const isOpen = !!q || open === s.id
            return (
              <div key={s.id} className="glass rounded-3xl border border-white/10 overflow-hidden">
                <button onClick={() => setOpen(isOpen ? null : s.id)} className="w-full flex items-center gap-3 p-5 text-left">
                  <div className="p-3 rounded-2xl bg-primary/10 text-primary"><Stethoscope className="h-5 w-5" /></div>
                  <div className="flex-1 min-w-0">
                    <div className="font-bold">{s.name}</div>
                    <div className="text-xs text-muted-foreground">{s.blurb}</div>
                  </div>
                  <Badge variant="outline">{cases.length} cases</Badge>
                  <ChevronRight className={`h-5 w-5 text-muted-foreground transition-transform ${isOpen ? "rotate-90" : ""}`} />
                </button>
                {isOpen && (
                  <div className="grid gap-2 sm:grid-cols-2 p-4 pt-0">
                    {cases.map((c) => (
                      <Link key={c.id} href={`/case-presentation/${c.id}/new`} className="rounded-2xl border border-white/10 p-4 hover:border-primary/50 transition-colors flex items-start gap-3">
                        <FileText className="h-5 w-5 text-primary mt-0.5 shrink-0" />
                        <div className="min-w-0">
                          <div className="font-semibold leading-tight">{c.title}</div>
                          <div className="text-xs text-muted-foreground mt-1">{c.blurb}</div>
                        </div>
                      </Link>
                    ))}
                  </div>
                )}
              </div>
            )
          })}
        </div>
        <p className="text-xs text-muted-foreground">{ALL_CASES.length} case types across {SUBJECTS.length} subjects.</p>
      </section>
    </div>
  )
}

"use client"

import { useCallback, useEffect, useMemo, useRef, useState } from "react"
import { useParams, useRouter } from "next/navigation"
import Link from "next/link"
import { Loader2 } from "lucide-react"
import { Button } from "@/components/ui/button"
import { UpgradeGate } from "@/components/upgrade-gate"
import { useFirestore } from "@/firebase"
import { getCase } from "@/lib/casepres"
import { sheetTitle } from "@/lib/casepres/format"
import { useCaseAccess, useCaseSheet, saveCaseSheet } from "@/hooks/use-case-sheets"
import { CaseForm } from "@/components/casepres/case-form"
import { CasePresentation } from "@/components/casepres/presentation"
import type { Answers } from "@/lib/casepres/types"

export default function CaseSheetPage() {
  const params = useParams<{ caseType: string; sheetId: string }>()
  const router = useRouter()
  const db = useFirestore()
  const caseDef = getCase(params.caseType)
  const { user, loading: accessLoading, allowed } = useCaseAccess()
  const initialId = params.sheetId === "new" ? null : params.sheetId
  const { sheet, loading: sheetLoading } = useCaseSheet(initialId)

  const [answers, setAnswers] = useState<Answers>({})
  const [mode, setMode] = useState<"edit" | "present">("edit")
  const [saveState, setSaveState] = useState<"idle" | "saving" | "saved" | "error">("idle")
  const idRef = useRef<string | null>(initialId)
  const loadedRef = useRef(false)
  const dirtyRef = useRef(false)

  const draftKey = `casepres-draft-${params.caseType}-${initialId ?? "new"}`

  // Load: saved sheet if opening one, otherwise a local draft for a new case.
  useEffect(() => {
    if (loadedRef.current) return
    if (initialId) {
      if (sheetLoading) return
      if (sheet) { setAnswers(sheet.answers || {}); loadedRef.current = true }
      else if (!sheetLoading) { loadedRef.current = true }
    } else {
      try {
        const raw = localStorage.getItem(draftKey)
        if (raw) setAnswers(JSON.parse(raw))
      } catch {}
      loadedRef.current = true
    }
  }, [initialId, sheet, sheetLoading, draftKey])

  const setField = useCallback((id: string, v: any) => {
    dirtyRef.current = true
    setAnswers((a) => ({ ...a, [id]: v }))
  }, [])

  // Autosave (debounced) to the student's account, with a local draft as a safety net.
  useEffect(() => {
    if (!dirtyRef.current || !caseDef || !db || !user || !allowed) return
    try { localStorage.setItem(draftKey, JSON.stringify(answers)) } catch {}
    setSaveState("saving")
    const t = setTimeout(async () => {
      try {
        const id = await saveCaseSheet(db, user.uid, idRef.current, {
          caseType: caseDef.id, subject: caseDef.subject, title: sheetTitle(caseDef, answers), answers,
        })
        if (!idRef.current) {
          idRef.current = id
          try { localStorage.removeItem(draftKey) } catch {}
          window.history.replaceState(null, "", `/case-presentation/${caseDef.id}/${id}`)
        }
        dirtyRef.current = false
        setSaveState("saved")
      } catch {
        setSaveState("error")
      }
    }, 1200)
    return () => clearTimeout(t)
  }, [answers, caseDef, db, user, allowed, draftKey])

  const content = useMemo(() => {
    if (!caseDef) return <div className="max-w-xl mx-auto p-10 text-center space-y-4"><p>That case type does not exist.</p><Link href="/case-presentation"><Button>Back to cases</Button></Link></div>
    if (accessLoading) return <div className="flex justify-center py-24"><Loader2 className="h-8 w-8 animate-spin text-primary" /></div>
    if (!user) return <div className="max-w-xl mx-auto p-10 text-center space-y-4"><p>Please sign in to use Case Presentation.</p><Button onClick={() => router.push("/login")}>Sign in</Button></div>
    if (!allowed) return <div className="max-w-xl mx-auto p-6 pt-12"><UpgradeGate type="content" title="Case Presentation is for subscribers" description="Subscribe to Scholar (₹999/year) to fill in case proformas and generate professional case presentations." /></div>
    if (initialId && sheetLoading) return <div className="flex justify-center py-24"><Loader2 className="h-8 w-8 animate-spin text-primary" /></div>
    if (mode === "present") return <CasePresentation caseDef={caseDef} answers={answers} onBack={() => setMode("edit")} />
    return <CaseForm caseDef={caseDef} answers={answers} setField={setField} onPresent={() => setMode("present")} saveState={saveState} />
  }, [caseDef, accessLoading, user, allowed, initialId, sheetLoading, mode, answers, saveState, setField, router])

  return content
}

function AddOwn({ placeholder, onAdd }: { placeholder: string; onAdd: (v: string) => void }) {
  const [v, setV] = useState("")
  const commit = () => {
    const t = v.trim()
    if (!t) return
    onAdd(t)
    setV("")
  }
  return (
    <div className="flex gap-2 mt-2">
      <Input
        value={v}
        onChange={(e) => setV(e.target.value)}
        onKeyDown={(e) => { if (e.key === "Enter") { e.preventDefault(); commit() } }}
        placeholder={placeholder}
        className="h-9 text-sm"
      />
      <button type="button" onClick={commit} className="h-9 px-3 rounded-md border border-white/15 text-sm hover:bg-white/5 flex items-center gap-1">
        <Plus className="h-4 w-4" /> Add
      </button>
    </div>
  )
}

/** Pick one or many from a list, or type your own. */
export function ChipPicker({ options, value, multi, onChange, allowCustom = true }: {
  options: string[]; value: string[]; multi?: boolean; onChange: (v: string[]) => void; allowCustom?: boolean
}) {
  const custom = value.filter((v) => !options.includes(v))
  const all = [...options, ...custom]
  const toggle = (o: string) => {
    if (multi) onChange(value.includes(o) ? value.filter((x) => x !== o) : [...value, o])
    else onChange(value[0] === o ? [] : [o])
  }
  return (
    <div>
      <div className="flex flex-wrap gap-2">
        {all.map((o) => {
          const on = value.includes(o)
          return (
            <button key={o} type="button" onClick={() => toggle(o)}
              className={cn(chipBase, on ? "bg-primary/20 border-primary text-primary font-medium" : "border-white/15 text-muted-foreground hover:border-white/30 hover:text-foreground")}>
              {on && <Check className="h-3.5 w-3.5 shrink-0" />}{o}
            </button>
          )
        })}
      </div>
      {allowCustom && (
        <AddOwn placeholder="Type your own and press Add" onAdd={(t) => { if (!value.includes(t)) onChange(multi ? [...value, t] : [t]) }} />
      )}
    </div>
  )
}

/** Tap once = present, twice = absent, thrice = clear. */
function FindingsPicker({ field, value, onChange, details, onDetails, photos, onPhotos, budget }: { field: Extract<Field, { t: "findings" }>; value: Record<string, "p" | "a">; onChange: (v: Record<string, "p" | "a">) => void; details: Record<string, string>; onDetails: (v: Record<string, string>) => void; photos: Record<string, string>; onPhotos: (v: Record<string, string>) => void; budget: number }) {
  const v = value || {}
  const custom = Object.keys(v).filter((k) => !field.options.includes(k))
  const all = [...field.options, ...custom]
  const cycle = (o: string) => {
    const next = { ...v }
    if (!v[o]) next[o] = "p"
    else if (v[o] === "p") next[o] = "a"
    else delete next[o]
    onChange(next)
  }
  const markRest = () => {
    const next = { ...v }
    all.forEach((o) => { if (!next[o]) next[o] = "a" })
    onChange(next)
  }
  const clear = () => onChange({})
  return (
    <div>
      <div className="flex flex-wrap gap-2">
        {all.map((o) => {
          const s = v[o]
          return (
            <button key={o} type="button" onClick={() => cycle(o)}
              className={cn(chipBase,
                s === "p" && "bg-primary/20 border-primary text-primary font-medium",
                s === "a" && "bg-white/5 border-white/10 text-muted-foreground line-through decoration-muted-foreground/60",
                !s && "border-white/15 text-muted-foreground hover:border-white/30 hover:text-foreground")}>
              {s === "p" && <Check className="h-3.5 w-3.5 shrink-0" />}
              {s === "a" && <Minus className="h-3.5 w-3.5 shrink-0" />}
              {o}
            </button>
          )
        })}
      </div>
      <div className="flex items-center gap-3 mt-2 text-xs text-muted-foreground flex-wrap">
        <span>Tap: <b className="text-primary">present</b> → <b>absent</b> → blank</span>
        <button type="button" onClick={markRest} className="underline hover:text-foreground">Mark rest as absent / normal</button>
        {Object.keys(v).length > 0 && <button type="button" onClick={clear} className="underline hover:text-foreground">Clear</button>}
      </div>
      <AddOwn placeholder="Add a finding that isn't listed (marked present)" onAdd={(t) => onChange({ ...v, [t]: "p" })} />
      <Elaborate value={v} details={details} onDetails={onDetails} photos={photos} onPhotos={onPhotos} budget={budget} />
    </div>
  )
}

/** For every present finding: explain it, and let the student describe site / extent. Tests show method, purpose and meaning. */
function resizePhoto(file: File): Promise<string> {
  return new Promise((resolve, reject) => {
    const img = new Image()
    const url = URL.createObjectURL(file)
    img.onload = () => {
      const max = 640
      const k = Math.min(1, max / Math.max(img.width, img.height))
      const cv = document.createElement("canvas")
      cv.width = Math.round(img.width * k); cv.height = Math.round(img.height * k)
      cv.getContext("2d")!.drawImage(img, 0, 0, cv.width, cv.height)
      URL.revokeObjectURL(url)
      let q = 0.7, out = cv.toDataURL("image/jpeg", q)
      while (out.length > 90000 && q > 0.3) { q -= 0.1; out = cv.toDataURL("image/jpeg", q) }
      resolve(out)
    }
    img.onerror = () => { URL.revokeObjectURL(url); reject(new Error("bad image")) }
    img.src = url
  })
}

        <input type="file" accept="image/*" className="hidden" onChange={async (e) => {
          const f = e.target.files?.[0]; e.target.value = ""
          if (!f) return
          try {
            const d = await resizePhoto(f)
            if (d.length > budget + (cur ? cur.length : 0)) { setErr("Photo limit reached. Remove another photo first."); return }
            setErr(""); onPhotos({ ...photos, [label]: d })
          } catch { setErr("Could not read that image.") }
        }} />
      </label>
      {cur && <button type="button" onClick={() => { const n = { ...photos }; delete n[label]; onPhotos(n) }} className="text-xs text-muted-foreground hover:text-foreground">Remove</button>}
      {err && <span className="text-xs text-red-400">{err}</span>}
    </div>
  )
}

export function FieldView({ field, answers, set }: { field: Field; answers: Answers; set: Setter }) {
  const v = answers[field.id]
  const wrap = (children: React.ReactNode, half = false) => (

      {field.t === "vitals" && <Label className="text-sm font-medium">{field.label}</Label>}
      {children}
    </div>
  )
  switch (field.t) {
    case "text":
      return wrap(<Input value={v || ""} onChange={(e) => set(field.id, e.target.value)} placeholder={field.placeholder} />, !!field.half)
    case "date":
      return wrap(<Input type="date" value={v || ""} onChange={(e) => set(field.id, e.target.value)} />, !!field.half)
    case "area":
      return wrap(<Textarea value={v || ""} onChange={(e) => set(field.id, e.target.value)} placeholder={field.placeholder} rows={3} />)
    case "choice":
      return wrap(<ChipPicker options={field.options} value={strArr(v)} multi={field.multi} onChange={(n) => set(field.id, field.multi ? n : n[0] ?? "")} />)
    case "findings":
      return wrap(<FindingsPicker field={field} value={v || {}} onChange={(n) => set(field.id, n)} details={answers[field.id + "__d"] || {}} onDetails={(n) => set(field.id + "__d", n)} photos={answers[field.id + "__p"] || {}} onPhotos={(n) => set(field.id + "__p", n)} budget={photoBudget(answers)} />)
    case "complaints":
      return wrap(<ComplaintsPicker field={field} value={v || {}} onChange={(n) => set(field.id, n)} />)
    case "symptom":
      return <div className="sm:col-span-2"><SymptomCard field={field} value={v || {}} onChange={(n) => set(field.id, n)} /></div>
    case "vitals":
      return wrap(<VitalsBlock field={field} value={v || {}} onChange={(n) => set(field.id, n)} />)
  }
}

const PHOTO_LIMIT = 700000
function photoBudget(answers: Record<string, any>) {
  let used = 0
  for (const k of Object.keys(answers)) if (k.endsWith("__p")) for (const v of Object.values(answers[k] || {})) used += String(v).length
  return Math.max(0, PHOTO_LIMIT - used)
}

export function CasePresentation({ caseDef, answers, onBack }: { caseDef: CaseDef; answers: Answers; onBack: () => void }) {
  const sections = useMemo(() => buildDocument(caseDef, answers), [caseDef, answers])
  const opening = useMemo(() => openingStatement(caseDef, answers), [caseDef, answers])
  const [slides, setSlides] = useState(false)
  const [i, setI] = useState(0)
  const [copied, setCopied] = useState(false)
  const [explain, setExplain] = useState(true)
  const today = new Date().toLocaleDateString("en-IN", { day: "numeric", month: "long", year: "numeric" })
  const total = sections.length + 1

  const next = useCallback(() => setI((x) => Math.min(x + 1, total - 1)), [total])
  const prev = useCallback(() => setI((x) => Math.max(x - 1, 0)), [])
  useEffect(() => {
    if (!slides) return
    const h = (e: KeyboardEvent) => {
      if (e.key === "ArrowRight" || e.key === " " || e.key === "PageDown") { e.preventDefault(); next() }
      else if (e.key === "ArrowLeft" || e.key === "PageUp") { e.preventDefault(); prev() }
      else if (e.key === "Escape") setSlides(false)
    }
    window.addEventListener("keydown", h)
    return () => window.removeEventListener("keydown", h)
  }, [slides, next, prev])

  const copy = async () => {
    try {
      await navigator.clipboard.writeText(docToText(caseDef, answers))
      setCopied(true)
      setTimeout(() => setCopied(false), 2000)
    } catch {}
  }

    <div className="max-w-4xl mx-auto px-4 py-6">

          Show test methods
        </label>
        <Button variant="outline" onClick={copy} className="gap-2">{copied ? <Check className="h-4 w-4" /> : <Copy className="h-4 w-4" />}{copied ? "Copied" : "Copy text"}</Button>
        <Button variant="outline" onClick={() => window.print()} className="gap-2"><Printer className="h-4 w-4" /> Print / PDF</Button>
